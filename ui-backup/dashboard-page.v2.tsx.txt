'use client';

import React, { useState, useEffect } from 'react';
import AppShell from '@/components/layout/AppShell';
import {
  AlertCircle,
  FileCheck,
  Clock,
  ShieldAlert,
  IndianRupee,
  Filter,
  CheckCircle2,
  Hourglass,
  TrendingUp,
  PackageCheck,
  Wallet,
  Receipt,
  CalendarDays,
  X,
  ChevronRight,
} from 'lucide-react';
import { formatINR, formatINRCompact, formatDateTime, getIndianFYQuarter, humanize, TIMEZONE } from '@/lib/utils';
import { formatQuotationNumber } from '@/lib/quotation';
import { DEFAULT_INDUSTRY_SEGMENTS } from '@/lib/industries';
import Link from 'next/link';
import DirectorQuoteApproveModal from '@/components/modals/DirectorQuoteApproveModal';
import { AlertTile, MetricCard, Panel, EmptyState, Avatar, FilterSelect, Skeleton } from '@/components/ui/dashboard';

function greeting() {
  const hour = Number(new Intl.DateTimeFormat('en-IN', { hour: 'numeric', hour12: false, timeZone: TIMEZONE }).format(new Date()));
  if (hour < 12) return 'Good morning';
  if (hour < 17) return 'Good afternoon';
  return 'Good evening';
}

function pct(part: number, whole: number) {
  if (!whole || whole <= 0) return 0;
  return Math.round((part / whole) * 100);
}

export default function DashboardPage() {
  const currentFY = getIndianFYQuarter();
  const [financialYear, setFinancialYear] = useState(currentFY.financialYear);
  const [quarter, setQuarter] = useState<string>(currentFY.quarter);
  const [salespersonId, setSalespersonId] = useState('');
  const [industry, setIndustry] = useState('');
  const [category, setCategory] = useState('');
  const [stage, setStage] = useState('');

  const [salespeople, setSalespeople] = useState<any[]>([]);
  const [stats, setStats] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  const [userName, setUserName] = useState('');

  const [industryOptions, setIndustryOptions] = useState<string[]>(DEFAULT_INDUSTRY_SEGMENTS);

  // Modals
  const [selectedQuoteForApproval, setSelectedQuoteForApproval] = useState<any>(null);

  useEffect(() => {
    fetch('/api/auth/me')
      .then((res) => (res.ok ? res.json() : null))
      .then((data) => setUserName(data?.user?.name || ''))
      .catch(() => {});
    fetch('/api/settings/users')
      .then((res) => res.json())
      .then((data) => {
        setSalespeople((data.users || []).filter((u: any) => u.role === 'SALESPERSON'));
      })
      .catch(() => {});
    fetch('/api/clients/industries')
      .then((res) => res.json())
      .then((data) => {
        if (data.industries) setIndustryOptions(data.industries);
      })
      .catch(() => {});
  }, []);

  async function loadDashboard() {
    setLoading(true);
    try {
      const params = new URLSearchParams({
        financialYear,
        quarter,
        ...(salespersonId ? { salespersonId } : {}),
        ...(industry ? { industry } : {}),
        ...(category ? { category } : {}),
        ...(stage ? { stage } : {}),
      });

      const res = await fetch(`/api/dashboard/stats?${params.toString()}`);
      if (res.ok) {
        const data = await res.json();
        setStats(data);
      }
    } catch {}
    setLoading(false);
  }

  useEffect(() => {
    loadDashboard();
    // Reload when a client is added from the sidebar
    window.addEventListener('crm:data-changed', loadDashboard);
    return () => window.removeEventListener('crm:data-changed', loadDashboard);
  }, [financialYear, quarter, salespersonId, industry, category, stage]);

  const isDirector = stats?.role === 'DIRECTOR';
  const summary = stats?.summary || {};
  const firstLoad = loading && !stats;
  const hasFilters = !!(salespersonId || industry || category || stage);

  const pipeline = Number(summary.totalPipelineValue || 0);
  const weighted = Number(summary.weightedPipelineValue || 0);
  const collected = Number(summary.totalPaymentsReceived || 0);
  const due = Number(summary.totalPaymentsPending || 0);
  const notYetDue = Number(summary.totalPaymentsNotYetDue || 0);

  const isCurrentQuarter = financialYear === currentFY.financialYear && quarter === currentFY.quarter;
  const daysLeft = Math.max(0, Math.ceil((currentFY.endDate.getTime() - Date.now()) / 86400000));
  const todayLabel = new Intl.DateTimeFormat('en-IN', { weekday: 'long', day: 'numeric', month: 'long', timeZone: TIMEZONE }).format(new Date());

  const actionCount =
    (summary.overdueTasksCount || 0) + (summary.dueTodayCount || 0) + (isDirector ? summary.pendingQuotesCount || 0 : 0);

  return (
    <AppShell>
      <div className="space-y-6">
        {/* Hero */}
        <section className="relative overflow-hidden rounded-3xl bg-gradient-to-br from-navy-900 via-navy-850 to-[#0c2a43] px-6 py-6 text-white shadow-lg shadow-navy-900/20 md:px-8 md:py-7">
          <div className="pointer-events-none absolute -right-16 -top-24 h-72 w-72 rounded-full bg-sky-500/20 blur-3xl" />
          <div className="pointer-events-none absolute -bottom-24 left-1/3 h-64 w-64 rounded-full bg-teal-400/10 blur-3xl" />
          <div
            className="pointer-events-none absolute inset-0 opacity-[0.07]"
            style={{ backgroundImage: 'radial-gradient(circle at 1px 1px, white 1px, transparent 0)', backgroundSize: '22px 22px' }}
          />
          <div className="relative flex flex-col gap-5 md:flex-row md:items-end md:justify-between">
            <div>
              <p className="flex items-center gap-1.5 text-xs font-medium text-sky-200/80">
                <CalendarDays className="h-3.5 w-3.5" />
                {todayLabel}
              </p>
              <h1 className="mt-2 text-2xl font-bold tracking-tight md:text-[28px]">
                {greeting()}
                {userName ? `, ${userName.split(' ')[0]}` : ''}
              </h1>
              <p className="mt-1.5 text-sm text-slate-300">
                {firstLoad ? (
                  'Loading your overview…'
                ) : actionCount > 0 ? (
                  <>
                    You have <span className="font-semibold text-white">{actionCount} item{actionCount === 1 ? '' : 's'}</span> needing
                    attention {isDirector ? 'across the company' : 'today'}.
                  </>
                ) : (
                  <>Everything is on track. {isDirector ? 'Whole company view.' : 'Your work view.'}</>
                )}
              </p>
            </div>
            <div className="flex flex-wrap items-center gap-2">
              <span className="inline-flex items-center gap-1.5 rounded-full border border-white/15 bg-white/10 px-3 py-1.5 text-xs font-semibold backdrop-blur">
                <span className="h-1.5 w-1.5 rounded-full bg-teal-300" />
                {financialYear} · {quarter}
              </span>
              {isCurrentQuarter && (
                <span className="inline-flex items-center rounded-full border border-white/15 bg-white/5 px-3 py-1.5 text-xs font-medium text-slate-300">
                  {daysLeft} days left in quarter
                </span>
              )}
              <span className="inline-flex items-center rounded-full border border-white/15 bg-white/5 px-3 py-1.5 text-xs font-medium text-slate-300">
                {isDirector ? 'Director view' : 'Salesperson view'}
              </span>
            </div>
          </div>
        </section>

        {/* Filters */}
        <div className="flex flex-wrap items-center gap-2">
          <span className="mr-1 inline-flex items-center gap-1.5 text-xs font-semibold text-slate-500">
            <Filter className="h-3.5 w-3.5" />
            Filters
          </span>
          <FilterSelect value={financialYear} onChange={setFinancialYear}>
            <option value="FY 2026-27">FY 2026-27</option>
            <option value="FY 2025-26">FY 2025-26</option>
          </FilterSelect>
          <FilterSelect value={quarter} onChange={setQuarter}>
            <option value="Q1">Q1 (Apr – Jun)</option>
            <option value="Q2">Q2 (Jul – Sep)</option>
            <option value="Q3">Q3 (Oct – Dec)</option>
            <option value="Q4">Q4 (Jan – Mar)</option>
          </FilterSelect>
          {isDirector && (
            <>
              <FilterSelect value={salespersonId} onChange={setSalespersonId} active={!!salespersonId}>
                <option value="">All salespeople</option>
                {salespeople.map((sp) => (
                  <option key={sp.id} value={sp.id}>
                    {sp.name}
                  </option>
                ))}
              </FilterSelect>
              <FilterSelect value={industry} onChange={setIndustry} active={!!industry}>
                <option value="">All industries</option>
                {industryOptions.map((name) => (
                  <option key={name} value={name}>
                    {name}
                  </option>
                ))}
                <option value="Other">Other</option>
              </FilterSelect>
            </>
          )}
          {hasFilters && (
            <button
              onClick={() => {
                setSalespersonId('');
                setIndustry('');
                setCategory('');
                setStage('');
              }}
              className="inline-flex items-center gap-1 rounded-full px-2.5 py-1.5 text-xs font-semibold text-slate-500 transition hover:bg-slate-100 hover:text-slate-700"
            >
              <X className="h-3.5 w-3.5" />
              Clear
            </button>
          )}
          {loading && stats && <span className="ml-auto h-4 w-4 animate-spin rounded-full border-2 border-slate-200 border-t-sky-500" />}
        </div>

        {/* 1. Needs action */}
        <div className="grid grid-cols-2 gap-3 md:gap-4 lg:grid-cols-4">
          <AlertTile
            href="/activities?view=overdue"
            label="Overdue follow-ups"
            value={summary.overdueTasksCount || 0}
            hint={summary.overdueTasksCount > 0 ? 'Needs action now' : 'All on time'}
            icon={ShieldAlert}
            tone={summary.overdueTasksCount > 0 ? 'rose' : 'slate'}
            pulse={summary.overdueTasksCount > 0}
            loading={firstLoad}
          />
          <AlertTile
            href="/activities?view=today"
            label="Due today"
            value={summary.dueTodayCount || 0}
            hint="Scheduled for today"
            icon={Clock}
            tone="sky"
            loading={firstLoad}
          />
          <AlertTile
            href="/proposals"
            label="Quotes to approve"
            value={summary.pendingQuotesCount || 0}
            hint={isDirector ? 'Waiting for you' : 'Waiting for director'}
            icon={Hourglass}
            tone="amber"
            loading={firstLoad}
          />
          <AlertTile
            href="/activities?view=missing_action"
            label="No next step"
            value={summary.missingActionCount || 0}
            hint="Requirements with no follow-up"
            icon={AlertCircle}
            tone={summary.missingActionCount > 0 ? 'amber' : 'emerald'}
            loading={firstLoad}
          />
        </div>

        {/* 2. Money */}
        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
          <MetricCard
            label="Pipeline"
            icon={TrendingUp}
            tone="sky"
            compact={formatINRCompact(pipeline)}
            exact={formatINR(pipeline)}
            loading={firstLoad}
            progress={{ value: pct(weighted, pipeline), label: `Likely ${formatINRCompact(weighted)} · ${pct(weighted, pipeline)}% weighted` }}
          />
          <MetricCard
            label="Orders booked"
            icon={PackageCheck}
            tone="emerald"
            compact={formatINRCompact(summary.totalOrdersNet)}
            exact={formatINR(summary.totalOrdersNet)}
            loading={firstLoad}
            footer={
              <div className="flex items-center justify-between">
                <span className="text-slate-500">With tax</span>
                <span className="font-bold tabular-nums text-slate-800">{formatINR(summary.totalOrdersGross)}</span>
              </div>
            }
          />
          <MetricCard
            label="Payments completed"
            icon={Wallet}
            tone="teal"
            compact={formatINRCompact(collected)}
            exact={formatINR(collected)}
            loading={firstLoad}
            progress={{ value: pct(collected, collected + due), label: `${pct(collected, collected + due)}% of raised dues collected` }}
          />
          <MetricCard
            label="Payment due"
            icon={Receipt}
            tone={due > 0 ? 'amber' : 'slate'}
            compact={formatINRCompact(due)}
            exact={formatINR(due)}
            loading={firstLoad}
            footer={
              <div className="flex items-center justify-between">
                <span className="text-slate-500">Not due yet</span>
                <span className="font-bold tabular-nums text-slate-800">{formatINR(notYetDue)}</span>
              </div>
            }
          />
        </div>

        {/* 3. Queues */}
        <div className="grid grid-cols-1 gap-6 lg:grid-cols-2">
          <Panel
            title="Overdue follow-ups"
            icon={ShieldAlert}
            iconClass="text-rose-500"
            count={stats?.overdueTasks?.length}
            countTone="rose"
            action={{ href: '/activities?view=overdue', label: 'View all' }}
          >
            {firstLoad ? (
              <QueueSkeleton />
            ) : !stats?.overdueTasks || stats.overdueTasks.length === 0 ? (
              <EmptyState icon={CheckCircle2} title="Nothing overdue" hint="Every follow-up is on schedule." />
            ) : (
              stats.overdueTasks.map((t: any) => (
                <Link
                  key={t.id}
                  href={`/clients/${t.clientId}`}
                  className="group flex items-center gap-3 px-5 py-3.5 transition hover:bg-slate-50"
                >
                  <Avatar name={t.client.name} />
                  <div className="min-w-0 flex-1">
                    <div className="flex items-center gap-2">
                      <span className="truncate text-sm font-semibold text-slate-900">{t.client.name}</span>
                      <span className="shrink-0 rounded-md bg-rose-50 px-1.5 py-0.5 text-[10px] font-bold text-rose-700 ring-1 ring-rose-100">
                        {humanize(t.actionType)}
                      </span>
                    </div>
                    <p className="mt-0.5 line-clamp-1 text-xs text-slate-500">{t.description}</p>
                    <p className="mt-1 text-[11px] text-slate-400">
                      Due <span className="font-semibold text-rose-600">{formatDateTime(t.dueDateTime)}</span>
                      {isDirector && <span> · {t.assignedSalesperson.name}</span>}
                    </p>
                  </div>
                  <ChevronRight className="h-4 w-4 shrink-0 text-slate-300 transition group-hover:translate-x-0.5 group-hover:text-sky-600" />
                </Link>
              ))
            )}
          </Panel>

          <Panel
            title={isDirector ? 'Needs your approval' : 'Waiting for approval'}
            icon={FileCheck}
            iconClass="text-sky-600"
            count={stats?.pendingQuotes?.length}
            countTone="amber"
            action={{ href: '/proposals', label: 'All quotations' }}
          >
            {firstLoad ? (
              <QueueSkeleton />
            ) : !stats?.pendingQuotes || stats.pendingQuotes.length === 0 ? (
              <EmptyState icon={CheckCircle2} title="Nothing waiting" hint="No quotations are pending approval." />
            ) : (
              stats.pendingQuotes.map((q: any) => (
                <div key={q.id} className="flex items-center gap-3 px-5 py-3.5 transition hover:bg-slate-50">
                  <Avatar name={q.client.name} />
                  <div className="min-w-0 flex-1">
                    <div className="flex items-center gap-2">
                      <span className="truncate text-sm font-semibold text-slate-900">{q.client.name}</span>
                      <span className="shrink-0 rounded-md bg-amber-50 px-1.5 py-0.5 text-[10px] font-bold text-amber-700 ring-1 ring-amber-100">
                        Discount
                      </span>
                    </div>
                    <p className="mt-0.5 text-xs text-slate-500">
                      {formatQuotationNumber(q.quotationNumber, q.currentVersionNumber)}
                      {isDirector && ` · ${q.salesperson.name}`}
                    </p>
                  </div>
                  <div className="flex shrink-0 flex-col items-end gap-1.5">
                    <span className="flex items-center text-sm font-bold tabular-nums text-slate-900">
                      <IndianRupee className="h-3.5 w-3.5" />
                      {formatINR(q.finalValue).replace('₹ ', '')}
                    </span>
                    {isDirector ? (
                      <button
                        onClick={() => setSelectedQuoteForApproval(q)}
                        className="rounded-lg bg-amber-500 px-3 py-1 text-xs font-semibold text-white shadow-sm shadow-amber-500/30 transition hover:bg-amber-600"
                      >
                        Review
                      </button>
                    ) : (
                      <span className="text-xs font-semibold text-amber-700">Pending</span>
                    )}
                  </div>
                </div>
              ))
            )}
          </Panel>
        </div>
      </div>

      {/* Modals */}
      {selectedQuoteForApproval && (
        <DirectorQuoteApproveModal
          isOpen={!!selectedQuoteForApproval}
          onClose={() => setSelectedQuoteForApproval(null)}
          onSuccess={() => {
            setSelectedQuoteForApproval(null);
            loadDashboard();
          }}
          quotation={selectedQuoteForApproval}
        />
      )}
    </AppShell>
  );
}

function QueueSkeleton() {
  return (
    <div>
      {[0, 1, 2].map((i) => (
        <div key={i} className="flex items-center gap-3 px-5 py-3.5">
          <Skeleton className="h-9 w-9 rounded-xl" />
          <div className="flex-1 space-y-2">
            <Skeleton className="h-3.5 w-1/2" />
            <Skeleton className="h-3 w-3/4" />
          </div>
        </div>
      ))}
    </div>
  );
}
