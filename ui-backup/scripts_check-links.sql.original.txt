-- Each row: a link rule and how many records break it (all should be 0)
select rule, broken from (
select 'Requirement number = client code / NNN' rule, count(*) broken from "Enquiry" e join "Client" c on c.id=e."clientId" where c.code is not null and e."enquiryNumber" like '%/%' and e."enquiryNumber" !~ ('^' || c.code || '/[0-9]{3}$')
union all select 'Client has a code', count(*) from "Client" where code is null
union all select 'Requirement stage = its last stage change', count(*) from "Enquiry" e where e.stage <> (select h."toStage" from "EnquiryStageHistory" h where h."enquiryId"=e.id order by h."createdAt" desc, h.id desc limit 1)
union all select 'Stage history chain is continuous', count(*) from (select h."fromStage", lag(h."toStage") over (partition by h."enquiryId" order by h."createdAt") prev from "EnquiryStageHistory" h) x where prev is not null and x."fromStage" <> prev
union all select 'Conversation client = its requirement client', count(*) from "Activity" a join "Enquiry" e on e.id=a."enquiryId" where e."clientId"<>a."clientId"
union all select 'Conversation contact belongs to the client', count(*) from "Activity" a join "Contact" ct on ct.id=a."contactId" where ct."clientId"<>a."clientId"
union all select 'Requirement contact/site belong to the client', count(*) from "Enquiry" e left join "Contact" ct on ct.id=e."contactId" left join "Site" s on s.id=e."siteId" where (ct.id is not null and ct."clientId"<>e."clientId") or (s.id is not null and s."clientId"<>e."clientId")
union all select 'Follow-up client = its requirement client', count(*) from "Task" t join "Enquiry" e on e.id=t."enquiryId" where e."clientId"<>t."clientId"
union all select 'Follow-up quotation/order/invoice = same client', count(*) from "Task" t left join "Quotation" q on q.id=t."quotationId" left join "Order" o on o.id=t."orderId" left join "Invoice" i on i.id=t."invoiceId" where (q.id is not null and q."clientId"<>t."clientId") or (o.id is not null and o."clientId"<>t."clientId") or (i.id is not null and i."clientId"<>t."clientId")
union all select 'Follow-up done <=> has outcome and completion time', count(*) from "Task" where (status='COMPLETED') <> ("completionOutcome" is not null and "completedAt" is not null)
union all select 'Follow-up assigned to a salesperson', count(*) from "Task" t join "User" u on u.id=t."assignedSalespersonId" where u.role<>'SALESPERSON'
union all select 'Quotation client = its requirement client', count(*) from "Quotation" q join "Enquiry" e on e.id=q."enquiryId" where e."clientId"<>q."clientId"
union all select 'Quotation current version = latest version', count(*) from "Quotation" q where q."currentVersionNumber" <> (select max(v."versionNumber") from "QuotationVersion" v where v."quotationId"=q.id)
union all select 'Quotation totals = latest version totals', count(*) from "Quotation" q join "QuotationVersion" v on v."quotationId"=q.id and v."versionNumber"=q."currentVersionNumber" where (q."originalValue",q."discountAmount",q."taxAmount",q."finalValue") <> (v."originalValue",v."discountAmount",v."taxAmount",v."finalValue")
union all select 'Quoted requirement value = its quotation price', count(*) from "Enquiry" e join lateral (select q."originalValue"-q."discountAmount" as price from "Quotation" q where q."enquiryId"=e.id order by q."createdAt" desc limit 1) lq on true where abs(e."estimatedValue" - lq.price) > 0.01
union all select 'Order payments add up to the order total (incl. GST)', count(*) from "Order" o where abs(o."grossValue" - coalesce((select sum(p.amount) from "OrderPaymentStage" p where p."orderId"=o.id),0)) > 0.01
union all select 'Order expected payment = total - TDS - retention', count(*) from "Order" where abs("expectedPayment" - ("grossValue" - "tdsAmount" - "retentionAmount")) > 0.01
union all select 'Stage expected amounts add up to the order expected payment', count(*) from "Order" o where abs(o."expectedPayment" - coalesce((select sum(p."expectedAmount") from "OrderPaymentStage" p where p."orderId"=o.id),0)) > 0.01
union all select 'Invoice step is a step of its own order', count(*) from "Order" o where o."invoiceStepId" is not null and not exists (select 1 from "OrderExecutionStep" s where s.id=o."invoiceStepId" and s."orderId"=o.id)
union all select 'Invoice GST is asked with exactly its order''s GST', count(*) from "Order" o where abs(o."invoiceGstAmount" - coalesce((select sum(p."gstAmount") from "OrderPaymentStage" p where p."orderId"=o.id),0)) > 0.01 or (o."invoiceStepId" is null and o."invoiceGstAmount" <> 0)
union all select 'Receipt is for a stage of its own order', count(*) from "OrderReceipt" r join "OrderPaymentStage" p on p.id=r."paymentStageId" where p."orderId" <> r."orderId"
union all select 'Receipt amount is positive', count(*) from "OrderReceipt" where amount <= 0
union all select 'Active payment reference is not repeated', count(*) from (select lower(reference) from "OrderReceipt" where reference is not null and not "isReversed" group by 1 having count(*) > 1) d
union all select 'Version totals add up (no GST, or 18% GST on older quotes)', count(*) from "QuotationVersion" v where (v."taxAmount" <> 0 and abs(v."taxAmount" - round((v."originalValue"-v."discountAmount")*0.18,2))>0.01) or abs(v."finalValue" - (v."originalValue"-v."discountAmount"+v."taxAmount"))>0.01
union all select 'Only the latest version is editable', count(*) from "QuotationVersion" v join "Quotation" q on q.id=v."quotationId" where v."isImmutable" = (v."versionNumber"=q."currentVersionNumber") and q."approvalStatus"<>'ACCEPTED'
union all select 'Waiting-for-approval quote has 1 pending request on latest version', count(*) from "Quotation" q where q."approvalStatus"='PENDING_APPROVAL' and (select count(*) from "QuotationApproval" a join "QuotationVersion" v on v.id=a."quotationVersionId" where a."quotationId"=q.id and a.status='PENDING' and v."versionNumber"=q."currentVersionNumber")<>1
union all select 'No pending request on a quote not waiting for approval', count(*) from "QuotationApproval" a join "Quotation" q on q.id=a."quotationId" where a.status='PENDING' and q."approvalStatus"<>'PENDING_APPROVAL'
union all select 'Order from quotation records its version', count(*) from "Order" where "quotationId" is not null and "quotationVersionId" is null
union all select 'Order client/requirement = quotation client/requirement', count(*) from "Order" o join "Quotation" q on q.id=o."quotationId" where q."clientId"<>o."clientId" or q."enquiryId" is distinct from o."enquiryId"
union all select 'Order version belongs to its quotation', count(*) from "Order" o join "QuotationVersion" v on v.id=o."quotationVersionId" where v."quotationId"<>o."quotationId"
union all select 'Order values = quotation version values', count(*) from "Order" o join "QuotationVersion" v on v.id=o."quotationVersionId" where o."grossValue"<>v."finalValue" or o."taxValue"<>v."taxAmount" or o."netValue"<>(v."originalValue"-v."discountAmount")
union all select 'Quotation with an order is Accepted', count(*) from "Order" o join "Quotation" q on q.id=o."quotationId" where q."approvalStatus"<>'ACCEPTED'
union all select 'Accepted quotation has an order', count(*) from "Quotation" q where q."approvalStatus"='ACCEPTED' and not exists (select 1 from "Order" o where o."quotationId"=q.id)
union all select 'Requirement with an order is Won', count(*) from "Order" o join "Enquiry" e on e.id=o."enquiryId" where e.stage<>'WON'
union all select 'Won requirement has a confirmed order', count(*) from "Enquiry" e where e.stage='WON' and not exists (select 1 from "Order" o where o."enquiryId"=e.id and o."orderStatus"<>'CANCELLED')
union all select 'Invoice client = its order client', count(*) from "Invoice" i join "Order" o on o.id=i."orderId" where i."clientId"<>o."clientId"
union all select 'Invoice totals add up (18% GST)', count(*) from "Invoice" where abs("grossAmount" - ("netAmount"+"taxAmount"))>0.01
union all select 'Invoices within order value', count(*) from "Order" o where (select coalesce(sum(i."grossAmount"),0) from "Invoice" i where i."orderId"=o.id and i."paymentStatus"<>'CANCELLED') > o."grossValue"+0.5
union all select 'Payment allocated to same-client invoices', count(*) from "ReceiptAllocation" ra join "Receipt" r on r.id=ra."receiptId" join "Invoice" i on i.id=ra."invoiceId" where r."clientId"<>i."clientId"
union all select 'Payment allocations within payment amount', count(*) from "Receipt" r where (select coalesce(sum("allocatedAmount"),0) from "ReceiptAllocation" where "receiptId"=r.id) > r.amount
union all select 'Invoice not over-allocated', count(*) from "Invoice" i where (select coalesce(sum(ra."allocatedAmount"),0) from "ReceiptAllocation" ra join "Receipt" r on r.id=ra."receiptId" where ra."invoiceId"=i.id and r."verificationStatus"<>'REJECTED') > i."grossAmount"+0.5
union all select 'Invoice status = verified payments', count(*) from "Invoice" i, lateral (select coalesce(sum(ra."allocatedAmount"),0) paid from "ReceiptAllocation" ra join "Receipt" r on r.id=ra."receiptId" where ra."invoiceId"=i.id and r."verificationStatus"='VERIFIED') p where i."paymentStatus"<>'CANCELLED' and i."paymentStatus"::text <> (case when p.paid>=i."grossAmount" then 'PAID' when p.paid>0 then 'PARTIALLY_PAID' else 'UNPAID' end)
union all select 'Verified payment has verifier and time', count(*) from "Receipt" where "verificationStatus"='VERIFIED' and ("verifiedById" is null or "verifiedAt" is null)
union all select 'Director note points at an existing record', count(*) from "DirectorComment" d where not exists (select 1 from "Client" where id=d."entityId") and not exists (select 1 from "Enquiry" where id=d."entityId") and not exists (select 1 from "Quotation" where id=d."entityId")
union all select 'Change-log entry points at an existing record', count(*) from "AuditEvent" a where not exists (select 1 from "Client" where id=a."entityId" union all select 1 from "Enquiry" where id=a."entityId" union all select 1 from "Quotation" where id=a."entityId" union all select 1 from "Order" where id=a."entityId" union all select 1 from "Invoice" where id=a."entityId" union all select 1 from "Receipt" where id=a."entityId" union all select 1 from "Task" where id=a."entityId") and not exists (select 1 from "AuditEvent" d where d."entityId"=a."entityId" and d.action in ('DELETE','PURGE')) and a."entityType" <> 'SETTINGS'
) checks;
